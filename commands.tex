% commands.tex — Reusable formatting macros
% Centralize all resume formatting here.

% --- Section ---
\newcommand{\ResumeSection}[1]{%
  \par\vspace{2pt}%
  \noindent\textbf{\MakeUppercase{#1}}%
  \par\vspace{1pt}%
  \noindent\rule{\textwidth}{0.4pt}%
  \par\vspace{2pt}%
}

% --- Lists ---
\newenvironment{ResumeItemize}{%
  \begin{itemize}[leftmargin=*,itemsep=0pt,topsep=0pt,parsep=0pt,partopsep=0pt,label=\textbullet]
}{%
  \end{itemize}
}

% --- Entry macros ---
\newcommand{\ExperienceEntry}[4]{%
  \noindent\textbf{#1} \hfill #2 \\
  \noindent\textit{#3} \hfill \textit{#4}%
  \vspace{1pt}%
}

\newcommand{\ProjectEntry}[4]{%
  \noindent\textbf{#1} \hfill #2 \\
  \noindent\textit{#3} \hfill \textit{#4}%
  \vspace{1pt}%
}

\newcommand{\AwardEntry}[3]{%
  \noindent\textbf{#1} \hfill #2 \\
  \noindent\textit{#3}\par%
  \vspace{1pt}%
}

\newcommand{\SkillCategory}[2]{%
  \noindent\textbf{#1:} #2\par%
  \vspace{1pt}%
}
